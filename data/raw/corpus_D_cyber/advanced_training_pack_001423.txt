#!/bin/bash
LOG_FILE="/var/log/security_monitor.log"
ALERT_FILE="/var/log/security_alerts.log"
log_message() { echo "[$(date '+%Y-%m-%d %H:%M:%S')] $1" | tee -a $LOG_FILE; }
alert_message() { echo "[$(date '+%Y-%m-%d %H:%M:%S')] ALERT: $1" | tee -a $ALERT_FILE; }
monitor_processes() { ps aux --no-headers | while read line; do pid=$(echo $line | awk '{print $2}'); cmd=$(echo $line | awk '{print $11}'); if echo $cmd | grep -qE "(nc|netcat|mimikatz)"; then alert_message "Suspicious tool: $cmd (PID: $pid)"; fi; done; }
monitor_network() { netstat -tulpn 2>/dev/null | grep LISTEN | while read line; do port=$(echo $line | awk '{print $4}' | cut -d: -f2); if [[ "$port" == "4444" ]] || [[ "$port" == "5555" ]]; then alert_message "Suspicious port: $port"; fi; done; }
log_message "Starting security monitoring..."; monitor_processes & monitor_network & wait

--- PYTHON SCRIPT: MEMORY FORENSICS ---
#!/usr/bin/env python3
import subprocess, sys
class MemoryAnalyzer:
    def __init__(self, dump): self.memory_dump = dump; self.profile = None
    def identify_profile(self):
        try:
            result = subprocess.run(['volatility', '-f', self.memory_dump, 'imageinfo'], capture_output=True, text=True)
            for line in result.stdout.split('\n'):
                if 'Suggested Profile' in line: self.profile = line.split(':')[1].strip().split(',')[0]; print(f"[+] Profile: {self.profile}"); return self.profile
        except Exception as e: print(f"[-] Error: {e}"); return None
    def list_processes(self):
        if not self.profile: print("[-] No profile"); return
        subprocess.run(['volatility', '-f', self.memory_dump, '--profile', self.profile, 'pslist'])
    def scan_network(self):
        if not self.profile: return
        subprocess.run(['volatility', '-f', self.memory_dump, '--profile', self.profile, 'netscan'])
    def find_malware(self):
        if not self.profile: return
        print("[*] Scanning for malware..."); subprocess.run(['volatility', '-f', self.memory_dump, '--profile', self.profile, 'malfind'])
if __name__ == "__main__":
    if len(sys.argv) < 2: print("Usage: python3 memory_forensics.py <memory_dump>"); sys.exit(1)
    analyzer = MemoryAnalyzer(sys.argv[1]); analyzer.identify_profile(); analyzer.list_processes(); analyzer.scan_network(); analyzer.find_malware()
